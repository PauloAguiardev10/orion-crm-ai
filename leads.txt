def valor_seguro(lead, coluna, padrao="NÃ£o informado"):
    try:
        valor = lead[coluna]
        if valor is None or str(valor).lower() in ["none", "nan", ""]:
            return padrao
        return valor
    except Exception:
        return padrao


def normalizar_status(status):
    status = str(status).strip().lower()

    if "aguardando" in status:
        return "Aguardando atendimento"

    if status == "em" or "em atendimento" in status:
        return "Em atendimento"

    if "proposta" in status:
        return "Proposta enviada"

    if "nÃ£o fechado" in status or "nao fechado" in status:
        return "NÃ£o fechado"

    if "fechado" in status:
        return "NegÃ³cio fechado"

    return "Aguardando atendimento"
